The corrected scan used the repository's already-installed `typescript-api` alias; it is reproduced below. No package change was needed.

```sh
node --input-type=module <<'NODE'
import { execFileSync } from 'node:child_process'
import { readFileSync } from 'node:fs'
import ts from 'typescript-api'
const paths = ['src/main/axiom', 'src/shared/axiom-contract', 'src/main/automations']
const changed = execFileSync('git', ['diff', '--name-only', '40b2bf0894^1', '40b2bf0894', '--', ...paths], { encoding: 'utf8' }).trim().split('\n')
const files = changed.filter(file => /\.(?:ts|tsx|js|mjs|cjs)$/.test(file) && !/\.(?:test|test-support)\./.test(file))
const findings = []
let regexCount = 0
let rawTemplateCount = 0
for (const file of files) {
  const source = ts.createSourceFile(file, readFileSync(file, 'utf8'), ts.ScriptTarget.Latest, true)
  function visit(node) {
    const isRegex = ts.isRegularExpressionLiteral(node)
    const isRawTemplate = ts.isTaggedTemplateExpression(node) && node.tag.getText(source) === 'String.raw'
    if (isRegex) regexCount += 1
    if (isRawTemplate) rawTemplateCount += 1
    if ((isRegex || isRawTemplate) && /[^\x00-\x7f]/.test(node.getText(source))) {
      const { line, character } = source.getLineAndCharacterOfPosition(node.getStart(source))
      findings.push({ file, line: line + 1, column: character + 1, kind: isRegex ? 'regex' : 'String.raw' })
    }
    ts.forEachChild(node, visit)
  }
  visit(source)
}
console.log(JSON.stringify({ scope: 'M3-02 attempts 1-3, current working tree', files, regexCount, rawTemplateCount, findings }, null, 2))
process.exitCode = findings.length ? 1 : 0
NODE
```
